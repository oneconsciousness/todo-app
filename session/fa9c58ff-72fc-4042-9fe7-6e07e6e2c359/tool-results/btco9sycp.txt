/** The task list screen: one ledger page with search, quick add, and the ruled list the user ticks off and reorders. */
export function TaskListView(props: TaskListViewProps) {
  const { content, today, query, searchPending, pendingCreate, failure } = props;
  const headingId = useId();
  const [draft, setDraft] = useState("");
  const [draftError, setDraftError] = useState<string | undefined>(undefined);
  const reorderable = !isSearching(query) && content.status === "ready";

  const handleSubmit = (event: SubmitEvent<HTMLFormElement>) => {
    event.preventDefault();
    const title = titleSchema.safeParse(draft);
    if (!title.success) {
      setDraftError(title.error.issues[0]?.message);
      return;
    }
    props.onCreate(title.data);
    setDraft("");
  };

  return (
    <Page labelledBy={headingId}>
      <header
        className={`flex flex-col gap-4 @wide:flex-row @wide:items-end @wide:justify-between ${inset}`}
      >
        <div className="flex flex-col gap-1">
          <p className="text-caption font-medium tracking-caps text-ink-muted uppercase">
            {longDate(today)}
          </p>
          <h1 id={headingId} className="font-display text-display font-medium text-balance">
            Tasks
          </h1>
          <p role="status" className="min-h-5 text-caption text-ink-muted tabular-nums">
            {content.status === "ready" && listSummary(content.tasks, query, searchPending)}
          </p>
        </div>
        <SearchBox
          label="Search tasks"
          placeholder="Search titles and notes"
          value={query}
          busy={searchPending}
          onChange={props.onQueryChange}
          className="@wide:w-1/2"
        />
      </header>

      <form noValidate onSubmit={handleSubmit} className={`flex items-start gap-2 ${inset}`}>
        <TextField
          label="New task"
          hideLabel
          placeholder="Add a task"
          value={draft}
          error={draftError}
          onChange={(event) => {
            setDraft(event.target.value);
            setDraftError(undefined);
          }}
          className="flex-1"
        />
        <Button type="submit" variant="primary" disabled={pendingCreate !== null}>
          <Icon name="plus" />
          Add task
        </Button>

        <div className={inset}>
          <Notice
            actions={failure.onRetry && [{ label: "Try again", onAction: failure.onRetry }]}

function ListBody({
  content,
  today,
  query,
  pendingCreate,
  reorderable,
  onQueryChange,
  onToggle,
  onMove,
  onRetry,
}: ListBodyProps) {
  switch (content.status) {
    case "loading":
      return <LoadingRows />;
    case "error":
      return (
        <EmptyState
          tone="alert"
          title="Couldn't load your tasks"
          actions={
            <Button variant="primary" onClick={onRetry}>
              Try again
            </Button>
          }
        >
          {content.message}
        </EmptyState>
      );
    case "ready":
      if (content.tasks.length === 0 && isSearching(query))
        return (
          <EmptyState
            title={`No tasks match “${query.trim()}”`}
            actions={
              <Button
                onClick={() => {
                  onQueryChange("");
                }}
              >
                Show all tasks
              </Button>
            }
          >
            Search looks through task titles and notes.
          </EmptyState>
        );
      if (content.tasks.length === 0 && pendingCreate === null)
        return (
          <EmptyState title="No tasks yet">
            Add your first task above. New tasks go to the top of the list.
          </EmptyState>
        );
      return (
        <SortableList
          items={content.tasks}
          label="Tasks"
          itemLabel={(task) => task.title}
          disabled={!reorderable}
          onMove={onMove}
          itemClassName={taskLineClass(reorderable)}
        >
          {(task, handle) => (
            <TaskRow task={task} today={today} handle={handle} onToggle={onToggle} />
          )}
        </SortableList>
      );
    default: {
      const unreachable: never = content;
      return unreachable;
    }
  }
}

interface Editing {
  /** The version the draft started from, or was last rebased onto. */
  readonly base: Task;
  readonly draft: TaskDraft;
  readonly errors: FieldErrors<TaskDraft>;
  /** The edit sent with the last save, so the view can recognise its own save when it comes back. */
  readonly submitted: TaskEdit | null;
}

interface TaskEntryProps extends Omit<TaskDetailViewProps, "content"> {
  readonly task: Task;
}

const isEmpty = (changes: TaskChanges): boolean => Object.keys(changes).length === 0;

function TaskEntry({
  task,
  today,
  saving,
  deleting,
  failure,
  onSave,
  onToggle,
  onDelete,
}: TaskEntryProps) {
  const headingId = useId();
  const [editing, setEditing] = useState<Editing | null>(null);
  const [confirmingDelete, setConfirmingDelete] = useState(false);
  const [seenVersion, setSeenVersion] = useState(task.version);

  // A newer version came in (React's pattern for adjusting state when a prop changes). If it is our own save,
  // leave edit mode; if it changes nothing the user can see, quietly rebase the draft onto it.
  if (task.version !== seenVersion) {
    setSeenVersion(task.version);
    if (editing?.submitted && isEmpty(changesFrom(task, editing.submitted))) setEditing(null);
    else if (editing && changedFieldNames(editing.base, task).length === 0)
      setEditing({ ...editing, base: task });
  }

  const startFrom = (latest: Task) => {
    setEditing({ base: latest, draft: draftFrom(latest), errors: {}, submitted: null });
  };

  const edit =
    (field: keyof TaskDraft) => (event: ChangeEvent<HTMLInputElement | HTMLTextAreaElement>) => {
      const { value } = event.target;
      setEditing(
        (current) =>
          current && {
            ...current,
            draft: { ...current.draft, [field]: value },
            errors: { ...current.errors, [field]: undefined },
            submitted: null,
          },
      );
    };

  const save = (event: SubmitEvent<HTMLFormElement>) => {
    event.preventDefault();
    if (!editing) return;
    const result = taskDraftSchema.safeParse(editing.draft);
    if (!result.success) {
      setEditing({ ...editing, errors: fieldErrors(result.error) });
      return;
    }
    const changes = changesFrom(editing.base, result.data);
    if (isEmpty(changes)) {
      setEditing(null);
      return;
    }
    onSave(task.id, changes, editing.base.version);
    setEditing({ ...editing, submitted: result.data });
  };

  const changedElsewhere =
    editing && task.version > editing.base.version ? changedFieldNames(editing.base, task) : null;
  const due = task.dueDate ? dueLabel(task.dueDate, today, task.completed) : null;

  return (
    <Page labelledBy={headingId}>
      <div className={inset}>
        <Link
          href={routes.tasks}
          className={`inline-flex min-h-touch items-center gap-2 transition-colors hover:text-ink ${captionCaps}`}
        >
          <Icon name="back" className="size-4" />
          All tasks
        </Link>
      </div>

      {failure && (
        <div className={inset}>
          <Notice>{failure}</Notice>
        </div>
      )}

      {editing ? (
        <form noValidate onSubmit={save} className={`flex flex-col gap-6 ${inset}`}>
          <h1 id={headingId} className="font-display text-title font-medium text-balance">
            Edit task
          </h1>
          {changedElsewhere && (
            <Notice
              actions={[
                {
                  label: "Use the latest",
                  onAction: () => {
                    startFrom(task);
                  },
                },
                {
                  label: "Keep my edits",
                  onAction: () => {
                    setEditing({ ...editing, base: task });
                  },
                },
              ]}
            >
              <strong className="font-semibold">Changed elsewhere.</strong> Another tab or device
              saved a newer version ({changedElsewhere.join(", ")}). Use the latest, or keep your
              edits and save over it.
            </Notice>
          )}
          <fieldset disabled={saving} className="flex min-w-0 flex-col gap-4">
            <TextField
              label="Title"
              value={editing.draft.title}
              error={editing.errors.title}
              onChange={edit("title")}
            />
            <TextField
              label="Due date"
              type="date"
              value={editing.draft.dueDate}
              error={editing.errors.dueDate}
              onChange={edit("dueDate")}
              className="@wide:w-1/2"
            />
            <TextArea
              label="Notes"
              rows={6}
              value={editing.draft.notes}
              error={editing.errors.notes}
              onChange={edit("notes")}
            />
          </fieldset>
          <div className="flex flex-wrap gap-2">
            <Button type="submit" variant="primary" disabled={saving}>
              {saving ? "Saving…" : "Save changes"}
            </Button>
            <Button
              disabled={saving}
              onClick={() => {
                setEditing(null);
              }}
            >
              Cancel
            </Button>
          </div>
        </form>
      ) : (
        <>
          <header className={`flex items-start gap-1 ${inset}`}>
            <span className="-ml-3">
              <Checkbox
                label={`Completed: ${task.title}`}
                checked={task.completed}
                onChange={(completed) => {
                  onToggle(task.id, completed);
                }}
              />
            </span>
            <div className="flex min-w-0 flex-col gap-2 pt-1">
              <h1
                id={headingId}
                className={`font-display text-display font-medium text-balance break-words ${task.completed ? "text-ink-muted line-through decoration-2" : ""}`}
              >
                {task.title}
              </h1>
              {due && (
                <p className="tabular-nums">
                  <time dateTime={task.dueDate ?? undefined} className={dueToneClass[due.tone]}>
                    {due.text}
                  </time>
                </p>
              )}
            </div>
          </header>

          <dl className={`grid grid-cols-3 gap-4 border-y border-rule py-4 ${inset}`}>
            {[
              ["Status", task.completed ? "Done" : "Open"],
              ["Entered", formatDay(task.createdAt, today)],

import { type ChangeEvent, type ReactNode, type SubmitEvent, useState } from "react";
import { type Credentials, credentialsSchema } from "../lib/schema/auth.ts";
import { type FieldErrors, fieldErrors } from "../lib/schema/field-errors.ts";
import { Button } from "./ui/button.tsx";
import { Notice } from "./ui/notice.tsx";
import { TextField } from "./ui/text-field.tsx";

/** What sign-in and sign-up both say when the server can't be reached. */
export const unreachableMessage = "Couldn't reach the server. Check your connection and try again.";

/** Props for `CredentialsForm`. */
export interface CredentialsFormProps {
  readonly submitLabel: string;
  readonly submittingLabel: string;
  readonly passwordAutoComplete: "current-password" | "new-password";
  readonly passwordHint?: string | undefined;
  readonly submitting: boolean;
  /** What went wrong on the last attempt, shown above the fields; null when nothing did. */
  readonly failure: ReactNode;
  readonly initialEmail?: string | undefined;
  readonly onSubmit: (credentials: Credentials) => void;
}

/** The email and password form behind sign-in and sign-up, validated by the same schema the API layer uses. */
export function CredentialsForm({
  submitLabel,
  submittingLabel,
  passwordAutoComplete,
  passwordHint,
  submitting,
  failure,
  initialEmail = "",
  onSubmit,
}: CredentialsFormProps) {
  const [values, setValues] = useState<Credentials>({ email: initialEmail, password: "" });
  const [errors, setErrors] = useState<FieldErrors<Credentials>>({});

  const edit = (field: keyof Credentials) => (event: ChangeEvent<HTMLInputElement>) => {
    setValues((current) => ({ ...current, [field]: event.target.value }));
    setErrors((current) => ({ ...current, [field]: undefined }));
  };

  const handleSubmit = (event: SubmitEvent<HTMLFormElement>) => {
    event.preventDefault();
    const result = credentialsSchema.safeParse(values);
    if (result.success) onSubmit(result.data);
    else setErrors(fieldErrors(result.error));
  };

  return (
    <form noValidate onSubmit={handleSubmit} className="flex flex-col gap-6">
      {failure && <Notice>{failure}</Notice>}
      <fieldset disabled={submitting} className="flex min-w-0 flex-col gap-4">
        <TextField
          label="Email"
          type="email"
          autoComplete="email"
          value={values.email}
          error={errors.email}
          onChange={edit("email")}
        />
        <TextField
          label="Password"
          type="password"
          autoComplete={passwordAutoComplete}
          hint={passwordHint}
          value={values.password}
          error={errors.password}
          onChange={edit("password")}
        />
      </fieldset>
      <Button type="submit" variant="primary" disabled={submitting} className="w-full">
        {submitting ? submittingLabel : submitLabel}
      </Button>
    </form>
  );
}
